
Task:Create a dataset with columns with the some missing values and salary missing values and cities( Hyderabad), experience with some duplicates, perform remove duplicates, handle missing values, standardize city names, apply min-max scaling on age and salary, show final clean data.

import pandas as pd
from sklearn.preprocessing import MinMaxScaler

# Create dataset
data = {
    "Name": ["Rahul", "Aisha", "John", "Rahul", "Meena"],
    "Age": [25, 30, None, 25, 28],
    "Salary": [50000, None, 45000, 50000, 60000],
    "City": ["Hyderabad", "hyderabad", "HYDERABAD", "Hyderabad", None],
    "Experience": [2, 5, 3, 2, 4]
}

df = pd.DataFrame(data)

print("Original Dataset:")
print(df)

# Remove duplicates
df = df.drop_duplicates()

# Handle missing values
df["Age"] = df["Age"].fillna(df["Age"].mean())
df["Salary"] = df["Salary"].fillna(df["Salary"].mean())
df["City"] = df["City"].fillna("Hyderabad")

# Standardize city names
df["City"] = df["City"].str.lower().str.title()

# Apply Min-Max Scaling
scaler = MinMaxScaler()

df[["Age", "Salary"]] = scaler.fit_transform(df[["Age", "Salary"]])

print("\nFinal Clean Data:")
print(df)
