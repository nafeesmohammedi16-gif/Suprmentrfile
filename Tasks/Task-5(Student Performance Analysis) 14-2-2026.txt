
Task: You are given a CSV file containing Name, Maths, Science, English, Department. 
     Requirement:- Read a CSV file, add new column ,Total, add a column Average, find top 3 students based on total department-wise average marks, students scoring above 75 in all subjects, and missing values if any.

import pandas as pd

# Read CSV file
df = pd.read_csv("students.csv")

# Display original data
print("Original Data:")
print(df)

# Check and fill missing values
df = df.fillna(0)

# Add Total column
df["Total"] = df["Maths"] + df["Science"] + df["English"]

# Add Average column
df["Average"] = df["Total"] / 3

# Top 3 students based on Total marks
top3 = df.sort_values(by="Total", ascending=False).head(3)

print("\nTop 3 Students:")
print(top3[["Name", "Total"]])

# Department-wise average marks
dept_avg = df.groupby("Department")[["Maths", "Science", "English"]].mean()

print("\nDepartment-wise Average Marks:")
print(dept_avg)

# Students scoring above 75 in all subjects
above75 = df[
    (df["Maths"] > 75) &
    (df["Science"] > 75) &
    (df["English"] > 75)
]

print("\nStudents Scoring Above 75 in All Subjects:")
print(above75[["Name", "Maths", "Science", "English"]])

# Display missing values count
print("\nMissing Values:")
print(df.isnull().sum())
