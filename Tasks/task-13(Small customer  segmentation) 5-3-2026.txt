  where create dataset on age, annual income, spending score, then create data frame, apply k-means and plot the clusters and display the centroid, (either 5 or 4 clusters).

# Customer Segmentation using K-Means

# Step 1: Import libraries
import pandas as pd
import matplotlib.pyplot as plt

from sklearn.cluster import KMeans

# Step 2: Create dataset

data = {
    'Age': [22, 25, 47, 52, 46, 56, 23, 27, 48, 55],
    
    'Annual_Income': [15000, 18000, 80000, 85000, 78000,
                      90000, 20000, 22000, 76000, 95000],
    
    'Spending_Score': [80, 75, 20, 18, 25,
                       15, 85, 70, 30, 10]
}

# Step 3: Create DataFrame

df = pd.DataFrame(data)

print("Customer Dataset:\n")
print(df)

# Step 4: Select features

X = df[['Annual_Income', 'Spending_Score']]

# Step 5: Apply K-Means Clustering

kmeans = KMeans(n_clusters=4, random_state=42)

kmeans.fit(X)

# Cluster labels
df['Cluster'] = kmeans.labels_

# Step 6: Print centroids

print("\nCentroids:\n")
print(kmeans.cluster_centers_)

# Step 7: Plot clusters

plt.scatter(
    df['Annual_Income'],
    df['Spending_Score'],
    c=df['Cluster'],
    s=100
)

# Plot centroids
plt.scatter(
    kmeans.cluster_centers_[:, 0],
    kmeans.cluster_centers_[:, 1],
    s=300,
    marker='X'
)

plt.xlabel("Annual Income")
plt.ylabel("Spending Score")

plt.title("Customer Segmentation using K-Means")

plt.show()
