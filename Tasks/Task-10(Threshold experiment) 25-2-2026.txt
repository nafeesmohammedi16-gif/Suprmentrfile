
Task: Threshold experiment using same dataset: Try thresholds 0.3, 0.5, 0.7. 
      Calculate accuracy, precision, recall. 
      Compare results in table. 
 Question: which threshold is best for cancer detection and why?

import pandas as pd
from sklearn.metrics import accuracy_score, precision_score, recall_score

# Actual values (1 = Cancer, 0 = No Cancer)
y_true = [1, 0, 1, 1, 0, 0, 1, 0, 1, 0]

# Predicted probabilities
y_prob = [0.9, 0.2, 0.8, 0.6, 0.4, 0.3, 0.7, 0.1, 0.85, 0.5]

# Thresholds
thresholds = [0.3, 0.5, 0.7]

# Store results
results = []

for t in thresholds:
    
    # Convert probabilities to predicted class
    y_pred = [1 if p >= t else 0 for p in y_prob]
    
    # Calculate metrics
    accuracy = accuracy_score(y_true, y_pred)
    precision = precision_score(y_true, y_pred)
    recall = recall_score(y_true, y_pred)
    
    # Store values
    results.append([t, accuracy, precision, recall])

# Create DataFrame
df = pd.DataFrame(results, columns=[
    "Threshold", "Accuracy", "Precision", "Recall"
])

# Print table
print(df)

# Best threshold for cancer detection
print("\nBest Threshold for Cancer Detection: 0.3")
print("Reason: Lower threshold gives higher recall,")
print("which helps detect more cancer cases and reduces missed patients.")
