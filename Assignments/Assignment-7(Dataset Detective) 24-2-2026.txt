import pandas as pd

# Step 1: Load dataset (example: CSV file)
df = pd.read_csv("sample_data.csv")

# Step 2: Display top 5 rows
print("Top 5 Rows:")
print(df.head())

# Step 3: Find column with highest value
max_values = df.max(numeric_only=True)
highest_column = max_values.idxmax()

print("\nColumn with Highest Maximum Value:", highest_column)
print("Highest Value:", max_values[highest_column])

# Step 4: Count missing values
print("\nMissing Values in Each Column:")
print(df.isnull().sum())
